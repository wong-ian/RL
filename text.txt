class MsPacManObjectExtractor:
    def __init__(self):
        # Calibrated colors based on your image
        self.colors = {
            'pacman': [210, 164, 74],      # Gold/Yellow
            'ghost_red': [200, 72, 72],    # Red Ghost
            'ghost_pink': [180, 122, 48],  # Pink Ghost
            'ghost_cyan': [84, 184, 153],  # Cyan Ghost
            'ghost_orange': [198, 108, 58],# Orange Ghost
            'ghost_blue': [66, 114, 194],  # Vulnerable Blue Ghost
            'fruit': [184, 70, 162],       # Cherry/Fruit
            
            # THE CRITICAL COLOR (Shared by Walls and Pellets)
            'wall_and_pellet': [228, 111, 111] 
        }
    
    def extract_objects(self, frame):
        # Crop to playable area (Remove top/bottom HUD)
        # Original is 210x160. Crop to 160x160.
        cropped = frame[25:185, :]
        
        # Resize to 40x40
        resized = cv2.resize(cropped, (40, 40), interpolation=cv2.INTER_NEAREST)
        
        # Initialize channels (11, 40, 40)
        channels = np.zeros((11, 40, 40), dtype=np.float32)
        
        # --- 1. Extract Simple Objects ---
        channels[0] = self._detect_color(resized, self.colors['pacman'])
        channels[1] = self._detect_color(resized, self.colors['ghost_red'])
        channels[2] = self._detect_color(resized, self.colors['ghost_pink'])
        channels[3] = self._detect_color(resized, self.colors['ghost_cyan'])
        channels[4] = self._detect_color(resized, self.colors['ghost_orange'])
        channels[5] = self._detect_color(resized, self.colors['ghost_blue'])
        channels[9] = self._detect_color(resized, self.colors['fruit'])
        
        # --- 2. Smart Pellet Detection (Color + Size Filter) ---
        # Step A: Detect EVERYTHING that is Pink (Walls + Pellets)
        pink_mask = self._detect_color(resized, self.colors['wall_and_pellet'], tolerance=50)
        
        # Step B: Filter out the big walls
        # We use connected components to find separate "blobs" of pink
        # Pellets are small blobs (1-2 pixels in 40x40). Walls are huge blobs.
        
        # Convert mask to uint8 for OpenCV
        pink_mask_uint8 = (pink_mask * 255).astype(np.uint8)
        
        # Find connected components
        num_labels, labels, stats, centroids = cv2.connectedComponentsWithStats(pink_mask_uint8, connectivity=8)
        
        pellet_channel = np.zeros_like(pink_mask)
        
        # Iterate through all detected pink blobs (start from 1 to skip background)
        for i in range(1, num_labels):
            area = stats[i, cv2.CC_STAT_AREA]
            
            # Heuristic: Pellets are small (e.g., < 6 pixels in 40x40 grid)
            # Walls are huge connected structures
            if area < 8: 
                # This is a pellet! Add it to the channel.
                pellet_channel[labels == i] = 1.0
                
        channels[10] = pellet_channel
        
        return channels

    def _detect_color(self, frame, target_color, tolerance=45):
        # Expanded tolerance to catch slight rendering variations
        diff = np.abs(frame.astype(np.int32) - np.array(target_color))
        mask = np.all(diff < tolerance, axis=2)
        return mask.astype(np.float32)
